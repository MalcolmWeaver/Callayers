cmake_minimum_required(VERSION 3.24)
project(Callayers LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Callayers requires Windows and the MSVC C++ toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A00000B)
add_compile_options(/W4 /permissive- /EHsc /utf-8)
add_library(camera_common INTERFACE)
target_include_directories(camera_common INTERFACE native)
target_link_libraries(camera_common INTERFACE mfplat mf mfuuid mfreadwrite ole32 oleaut32 propsys advapi32 shlwapi mfsensorgroup ksuser windowscodecs crypt32 shell32)
add_library(CallayersCamera SHARED native/source.cpp native/source.def)
target_link_libraries(CallayersCamera PRIVATE camera_common)
add_executable(Callayers WIN32 native/app.cpp native/app.manifest native/app.rc)
if(CALLAYERS_APP_NAME)
  set_target_properties(Callayers PROPERTIES OUTPUT_NAME "${CALLAYERS_APP_NAME}")
endif()
target_compile_options(Callayers PRIVATE /bigobj)
target_link_options(Callayers PRIVATE /DELAYLOAD:Microsoft.UI.Windowing.Core.dll)
if(NOT CALLAYERS_WINAPPSDK)
  set(CALLAYERS_WINAPPSDK "$ENV{USERPROFILE}/.nuget/packages/microsoft.windowsappsdk/1.6.250602001")
endif()
target_include_directories(Callayers BEFORE PRIVATE "${CMAKE_BINARY_DIR}/generated" "${CALLAYERS_WINAPPSDK}/include")
target_link_libraries(Callayers PRIVATE camera_common gdi32 user32 comctl32 comdlg32 d2d1 dwrite windowscodecs runtimeobject dwmapi delayimp
  "${CALLAYERS_WINAPPSDK}/lib/win10-x64/Microsoft.WindowsAppRuntime.Bootstrap.lib"
  "${CALLAYERS_WINAPPSDK}/lib/native/win10-x64/Microsoft.UI.Dispatching.lib")
add_custom_command(TARGET Callayers POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${CALLAYERS_WINAPPSDK}/runtimes/win-x64/native/Microsoft.WindowsAppRuntime.Bootstrap.dll" "$<TARGET_FILE_DIR:Callayers>")
add_executable(EditorTests native/editor_tests.cpp)
target_link_libraries(EditorTests PRIVATE camera_common d2d1 dwrite windowscodecs runtimeobject)
add_executable(EditorUITests native/editor_ui_tests.cpp)
target_link_libraries(EditorUITests PRIVATE camera_common d2d1 dwrite windowscodecs runtimeobject gdi32 user32 comctl32 comdlg32)
add_executable(CameraProbe native/probe.cpp)
target_link_libraries(CameraProbe PRIVATE camera_common)
add_library(CallayersCameraTest SHARED native/source.cpp native/source.def)
target_compile_definitions(CallayersCameraTest PRIVATE CALLAYERS_DEMAND_TEST_PIPE)
target_link_libraries(CallayersCameraTest PRIVATE camera_common)
enable_testing()
add_executable(CameraDemandTests native/camera_demand_tests.cpp)
target_link_libraries(CameraDemandTests PRIVATE camera_common)
add_test(NAME camera_demand COMMAND CameraDemandTests)
add_test(NAME frame_protocol COMMAND CameraProbe --self-test)
add_test(NAME source_lifecycle COMMAND CameraProbe --source-test $<TARGET_FILE:CallayersCameraTest>)
add_test(NAME editor_document COMMAND EditorTests)
add_test(NAME editor_interactions COMMAND EditorUITests)
add_test(NAME shell_editor_commands COMMAND EditorUITests --modern)
add_executable(SetupTests native/setup_tests.cpp)
target_link_libraries(SetupTests PRIVATE camera_common d2d1 dwrite windowscodecs runtimeobject gdi32 user32 comctl32 comdlg32)
add_test(NAME setup_flow COMMAND SetupTests)
add_executable(ImageTests native/image_tests.cpp)
target_link_libraries(ImageTests PRIVATE camera_common d2d1 dwrite runtimeobject gdi32 user32 comctl32 comdlg32)
add_test(NAME local_images COMMAND ImageTests)

if(NOT CALLAYERS_WEBVIEW2)
  set(CALLAYERS_WEBVIEW2 "$ENV{USERPROFILE}/.nuget/packages/microsoft.web.webview2/1.0.2651.64")
  if(NOT EXISTS "${CALLAYERS_WEBVIEW2}/build/native/include/WebView2.h")
    set(CALLAYERS_WEBVIEW2 "${CMAKE_BINARY_DIR}/packages/microsoft.web.webview2/1.0.2651.64")
  endif()
endif()
add_library(card_renderer STATIC native/card_host.cpp)
target_include_directories(card_renderer PRIVATE "${CALLAYERS_WEBVIEW2}/build/native/include")
target_link_libraries(card_renderer PUBLIC camera_common d3d11 CoreMessaging windowsapp d2d1 dwrite windowscodecs runtimeobject shell32
  "${CALLAYERS_WEBVIEW2}/build/native/x64/WebView2LoaderStatic.lib")
add_executable(CardProbe native/card_probe.cpp)
target_link_libraries(CardProbe PRIVATE card_renderer)

target_link_libraries(Callayers PRIVATE card_renderer)

set(CALLAYERS_SERVICE_URL "" CACHE STRING "HTTPS Cloud Run origin for Callayers API")
add_library(ai_client STATIC native/ai_client.cpp native/website.cpp native/cloud_client.cpp)
target_compile_definitions(ai_client PRIVATE CALLAYERS_SERVICE_URL=L"${CALLAYERS_SERVICE_URL}")
target_link_libraries(ai_client PUBLIC camera_common winhttp shell32 runtimeobject ws2_32 bcrypt)
target_link_libraries(Callayers PRIVATE ai_client)
add_executable(AiTests native/ai_tests.cpp)
target_link_libraries(AiTests PRIVATE ai_client)
add_test(NAME ai_generation_contract COMMAND AiTests)
add_test(NAME website_worker_lifecycle COMMAND AiTests --website-worker-test)
add_executable(WebsiteMediaTests native/website_media_tests.cpp)
target_link_libraries(WebsiteMediaTests PRIVATE ai_client)
add_test(NAME website_media COMMAND WebsiteMediaTests)

add_executable(AiProbe native/ai_probe.cpp)
target_link_libraries(AiProbe PRIVATE ai_client)

cmake_minimum_required(VERSION 3.24)
project(CallayersFoundation VERSION 0.2.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The foundation requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CALLAYERS_ULTRALIGHT_CPU "Use isolated Ultralight CPU overlay" OFF)
set(WEBVIEW2_ROOT "$ENV{USERPROFILE}/.nuget/packages/microsoft.web.webview2/1.0.2651.64/build/native" CACHE PATH "WebView2 SDK native folder")
if(NOT EXISTS "${WEBVIEW2_ROOT}/include/WebView2.h")
  message(FATAL_ERROR "WebView2 SDK missing. Run scripts/build.ps1 to acquire the pinned SDK.")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /EHsc /utf-8)
set(WINDOWSAPPSDK_ROOT "$ENV{USERPROFILE}/.nuget/packages/microsoft.windowsappsdk/1.6.250602001" CACHE PATH "Windows App SDK native package")
if(NOT EXISTS "${WINDOWSAPPSDK_ROOT}/include/MddBootstrap.h")
  message(FATAL_ERROR "Windows App SDK package missing. Run scripts/build.ps1.")
endif()
set(WINUI_PROJECTIONS "${CMAKE_BINARY_DIR}/generated")
set(SDK_TOOLS "$ENV{ProgramFiles\(x86\)}/Windows Kits/10")
find_program(CPPWINRT_EXE cppwinrt.exe PATHS "${SDK_TOOLS}/bin/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/x64" REQUIRED NO_DEFAULT_PATH)
get_filename_component(WEBVIEW2_PACKAGE "${WEBVIEW2_ROOT}/../.." ABSOLUTE)
if(NOT EXISTS "${WINUI_PROJECTIONS}/.complete-v2")
  execute_process(COMMAND "${CPPWINRT_EXE}" -input "${WINDOWSAPPSDK_ROOT}/lib/uap10.0" -input "${WINDOWSAPPSDK_ROOT}/lib/uap10.0.18362"
    -reference "${SDK_TOOLS}/UnionMetadata/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/Windows.winmd" -input "${WEBVIEW2_PACKAGE}/lib/Microsoft.Web.WebView2.Core.winmd" -output "${WINUI_PROJECTIONS}" RESULT_VARIABLE projection_result)
  if(NOT projection_result EQUAL 0)
    message(FATAL_ERROR "Could not generate WinUI C++ projections")
  endif()
  file(WRITE "${WINUI_PROJECTIONS}/.complete-v2" "Windows App SDK 1.6.250602001 and WebView2 projections generated successfully")
endif()
set(OVERLAY_SOURCE native/overlay.cpp)
if(CALLAYERS_ULTRALIGHT_CPU)
  set(OVERLAY_SOURCE native/overlay_ultralight.cpp)
endif()
add_executable(Callayers WIN32 native/app.cpp native/auth.cpp native/auth_protocol.cpp native/auth_store.cpp native/capture.cpp native/camera_preferences.cpp native/annotations.cpp native/annotation_store.cpp ${OVERLAY_SOURCE} native/text_editor.cpp native/workspace.cpp native/app.rc)
if(CALLAYERS_ULTRALIGHT_CPU)
  set(ULTRALIGHT_ROOT "${CMAKE_SOURCE_DIR}/.deps/ultralight" CACHE PATH "Ultralight SDK")
  target_compile_definitions(Callayers PRIVATE CALLAYERS_ULTRALIGHT_CPU)
  target_include_directories(Callayers SYSTEM PRIVATE "${ULTRALIGHT_ROOT}/include")
  target_link_directories(Callayers PRIVATE "${ULTRALIGHT_ROOT}/lib")
  target_link_libraries(Callayers PRIVATE Ultralight UltralightCore WebCore AppCore)
  add_custom_command(TARGET Callayers POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${ULTRALIGHT_ROOT}/bin" "$<TARGET_FILE_DIR:Callayers>"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${ULTRALIGHT_ROOT}/resources" "$<TARGET_FILE_DIR:Callayers>/resources"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${ULTRALIGHT_ROOT}/license" "$<TARGET_FILE_DIR:Callayers>/ultralight-license")
endif()
target_include_directories(Callayers PRIVATE native "${WINUI_PROJECTIONS}" "${WEBVIEW2_ROOT}/include" "${WINDOWSAPPSDK_ROOT}/include")
target_link_options(Callayers PRIVATE /MANIFEST:NO /DELAYLOAD:Microsoft.UI.Windowing.Core.dll)
target_link_libraries(Callayers PRIVATE delayimp winhttp ws2_32 bcrypt crypt32)
# ContentPreTranslateMessage is imported from the runtime package. Resolve it
# only after MddBootstrapInitialize2 has added that package to DLL search paths.
target_link_libraries(Callayers PRIVATE "${WEBVIEW2_ROOT}/x64/WebView2LoaderStatic.lib" "${WINDOWSAPPSDK_ROOT}/lib/win10-x64/Microsoft.WindowsAppRuntime.Bootstrap.lib" "${WINDOWSAPPSDK_ROOT}/lib/native/win10-x64/Microsoft.UI.Dispatching.lib" mfplat mfreadwrite mfuuid mf windowscodecs ole32 oleaut32 shlwapi shell32 comctl32 gdi32 user32 advapi32 windowsapp dwmapi)
add_library(CallayersCamera SHARED native/camera_filter.cpp native/camera.def)
target_include_directories(CallayersCamera PRIVATE native)
target_link_libraries(CallayersCamera PRIVATE strmiids ole32 oleaut32 advapi32 user32 gdi32 winmm)
include(CTest)
if(BUILD_TESTING)
  add_executable(AuthProtocolTests tests/auth_protocol_test.cpp native/auth_protocol.cpp)
  target_include_directories(AuthProtocolTests PRIVATE native)
  target_link_libraries(AuthProtocolTests PRIVATE bcrypt crypt32)
  add_test(NAME authentication_protocol_and_protection COMMAND AuthProtocolTests)
  add_executable(AnnotationTests tests/annotations_test.cpp native/annotations.cpp)
  target_include_directories(AnnotationTests PRIVATE native "${WINUI_PROJECTIONS}")
  target_link_libraries(AnnotationTests PRIVATE windowsapp ole32)
  add_test(NAME annotation_geometry_and_restore COMMAND AnnotationTests)
  find_program(NODE_EXECUTABLE node)
  if(NODE_EXECUTABLE)
    add_test(NAME text_editor_enter_and_cancel COMMAND "${NODE_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/text_editor_script_test.js" "${CMAKE_SOURCE_DIR}/web/text-editor.html")
  endif()
endif()
add_custom_command(TARGET Callayers POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:Callayers>/web")
add_custom_command(TARGET Callayers POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WINDOWSAPPSDK_ROOT}/runtimes/win-x64/native/Microsoft.WindowsAppRuntime.Bootstrap.dll" "$<TARGET_FILE_DIR:Callayers>")
add_custom_command(TARGET Callayers POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/spec-v2/assets" "$<TARGET_FILE_DIR:Callayers>/assets")

cmake_minimum_required(VERSION 3.24)
set(CALLAYERS_VERSION "0.3.0" CACHE STRING "Desktop release version")
project(CallayersFoundation VERSION ${CALLAYERS_VERSION} LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The foundation requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(CALLAYERS_INTERACTIVE "Enable fixed interactive card input/state" OFF)
option(CALLAYERS_GENERATION "Enable AI creation and saved generated cards" OFF)
option(CALLAYERS_IMAGES "Enable local image import" OFF)
option(CALLAYERS_LIBRARY "Enable saved library management and revision" OFF)
option(CALLAYERS_TELEMETRY "Enable MVP usage telemetry and consumer identity" OFF)
option(CALLAYERS_RELEASE "Build installable release with automatic updates" OFF)
if(CALLAYERS_RELEASE AND NOT CALLAYERS_TELEMETRY)
  message(FATAL_ERROR "Release requires all accepted MVP features")
endif()
if(CALLAYERS_TELEMETRY AND NOT CALLAYERS_LIBRARY)
  message(FATAL_ERROR "Telemetry build requires the current saved library workspace")
endif()
if(CALLAYERS_LIBRARY AND NOT CALLAYERS_IMAGES)
  message(FATAL_ERROR "Library revision requires image support")
endif()
if(CALLAYERS_IMAGES AND NOT CALLAYERS_GENERATION)
  message(FATAL_ERROR "Image library requires saved cards")
endif()
# WinUI metadata references its optional browser control. Resolve those types only
# while generating projections; Callayers does not link or instantiate WebView2.
set(WEBVIEW2_METADATA "" CACHE FILEPATH "Reference metadata for Windows App SDK projection")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /EHsc /utf-8)
set(WINDOWSAPPSDK_ROOT "$ENV{USERPROFILE}/.nuget/packages/microsoft.windowsappsdk/1.6.250602001" CACHE PATH "Windows App SDK native package")
if(NOT EXISTS "${WINDOWSAPPSDK_ROOT}/include/MddBootstrap.h")
  message(FATAL_ERROR "Windows App SDK package missing. Run scripts/build.ps1.")
endif()
set(WINUI_PROJECTIONS "${CMAKE_BINARY_DIR}/generated")
set(SDK_TOOLS "$ENV{ProgramFiles\(x86\)}/Windows Kits/10")
find_program(CPPWINRT_EXE cppwinrt.exe PATHS "${SDK_TOOLS}/bin/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/x64" REQUIRED NO_DEFAULT_PATH)
if(NOT EXISTS "${WINUI_PROJECTIONS}/.complete-ultralight-editor-v2")
  execute_process(COMMAND "${CPPWINRT_EXE}" -input "${WINDOWSAPPSDK_ROOT}/lib/uap10.0" -input "${WINDOWSAPPSDK_ROOT}/lib/uap10.0.18362"
    -reference "${SDK_TOOLS}/UnionMetadata/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/Windows.winmd" -input "${WEBVIEW2_METADATA}" -output "${WINUI_PROJECTIONS}" RESULT_VARIABLE projection_result)
  if(NOT projection_result EQUAL 0)
    message(FATAL_ERROR "Could not generate WinUI C++ projections")
  endif()
  file(WRITE "${WINUI_PROJECTIONS}/.complete-ultralight-editor-v2" "Windows App SDK 1.6.250602001 projections generated successfully")
endif()
add_executable(Callayers WIN32 native/app.cpp native/auth.cpp native/auth_protocol.cpp native/auth_store.cpp native/capture.cpp native/camera_preferences.cpp native/annotations.cpp native/annotation_store.cpp native/overlay_ultralight.cpp native/text_editor.cpp native/workspace.cpp native/app.rc)
configure_file(release/version.rc.in "${CMAKE_BINARY_DIR}/release-version.rc" @ONLY)
target_sources(Callayers PRIVATE "${CMAKE_BINARY_DIR}/release-version.rc")
target_compile_definitions(Callayers PRIVATE CALLAYERS_VERSION_TEXT=L"${PROJECT_VERSION}")
if(CALLAYERS_RELEASE)
  target_compile_definitions(Callayers PRIVATE CALLAYERS_RELEASE)
  add_executable(CallayersLauncher WIN32 native/release_launcher.cpp native/app.rc "${CMAKE_BINARY_DIR}/release-version.rc")
  target_link_options(CallayersLauncher PRIVATE /MANIFEST:NO)
  set_property(TARGET CallayersLauncher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(CallayersLauncher PRIVATE user32)
endif()
if(CALLAYERS_INTERACTIVE)
  target_compile_definitions(Callayers PRIVATE CALLAYERS_INTERACTIVE)
  target_sources(Callayers PRIVATE native/card_state.cpp)
endif()
if(CALLAYERS_GENERATION)
  target_compile_definitions(Callayers PRIVATE CALLAYERS_GENERATION)
  target_sources(Callayers PRIVATE native/generated_card.cpp native/generation.cpp)
endif()
if(CALLAYERS_IMAGES)
  target_compile_definitions(Callayers PRIVATE CALLAYERS_IMAGES)
  target_sources(Callayers PRIVATE native/image_import.cpp)
endif()
if(CALLAYERS_LIBRARY)
  target_compile_definitions(Callayers PRIVATE CALLAYERS_LIBRARY)
endif()
set(ULTRALIGHT_ROOT "${CMAKE_SOURCE_DIR}/.deps/ultralight" CACHE PATH "Ultralight SDK")
target_include_directories(Callayers SYSTEM PRIVATE "${ULTRALIGHT_ROOT}/include")
target_link_directories(Callayers PRIVATE "${ULTRALIGHT_ROOT}/lib")
target_link_libraries(Callayers PRIVATE Ultralight UltralightCore WebCore AppCore)
add_custom_command(TARGET Callayers POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${ULTRALIGHT_ROOT}/bin" "$<TARGET_FILE_DIR:Callayers>"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${ULTRALIGHT_ROOT}/resources" "$<TARGET_FILE_DIR:Callayers>/resources"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${ULTRALIGHT_ROOT}/license" "$<TARGET_FILE_DIR:Callayers>/ultralight-license")

target_include_directories(Callayers PRIVATE native "${WINUI_PROJECTIONS}" "${WINDOWSAPPSDK_ROOT}/include")
target_link_options(Callayers PRIVATE /MANIFEST:NO /DELAYLOAD:Microsoft.UI.Windowing.Core.dll)
target_link_libraries(Callayers PRIVATE delayimp winhttp ws2_32 bcrypt crypt32)
# ContentPreTranslateMessage is imported from the runtime package. Resolve it
# only after MddBootstrapInitialize2 has added that package to DLL search paths.
target_link_libraries(Callayers PRIVATE "${WINDOWSAPPSDK_ROOT}/lib/win10-x64/Microsoft.WindowsAppRuntime.Bootstrap.lib" "${WINDOWSAPPSDK_ROOT}/lib/native/win10-x64/Microsoft.UI.Dispatching.lib" mfplat mfreadwrite mfuuid mf windowscodecs ole32 oleaut32 shlwapi shell32 comctl32 gdi32 user32 advapi32 windowsapp dwmapi msimg32 imm32)
add_library(CallayersCamera SHARED native/camera_filter.cpp native/camera.def)
if(CALLAYERS_TELEMETRY)
  target_compile_definitions(Callayers PRIVATE CALLAYERS_TELEMETRY)
  target_compile_definitions(CallayersCamera PRIVATE CALLAYERS_TELEMETRY)
  target_sources(Callayers PRIVATE native/telemetry.cpp native/telemetry_state.cpp)
endif()

target_link_libraries(CallayersCamera PRIVATE strmiids ole32 oleaut32 advapi32 user32 gdi32 winmm)

cmake_minimum_required(VERSION 3.22)
set(CALLAYERS_RELEASE_VERSION "0.3.5" CACHE STRING "Numeric Linux release version")
if(NOT CALLAYERS_RELEASE_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "Release version must be major.minor.patch")
endif()
project(CallayersLinux VERSION ${CALLAYERS_RELEASE_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(WIN32)
  message(FATAL_ERROR "The Linux app is built on Linux. The Windows app uses the repository root.")
endif()
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network DBus)
set(ULTRALIGHT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/ultralight-linux" CACHE PATH "Linux Ultralight 2.0.0-beta.2 SDK")
if(NOT EXISTS "${ULTRALIGHT_ROOT}/include/Ultralight/Ultralight.h")
  message(FATAL_ERROR "Linux Ultralight SDK missing. Run python3 linux/packaging/acquire-renderer.py")
endif()
file(READ "${ULTRALIGHT_ROOT}/VERSION.txt" ULTRALIGHT_VERSION)
string(STRIP "${ULTRALIGHT_VERSION}" ULTRALIGHT_VERSION)
if(NOT ULTRALIGHT_VERSION STREQUAL "2.0.0-beta.2")
  message(FATAL_ERROR "Expected Ultralight 2.0.0-beta.2")
endif()
add_library(callayers_renderer STATIC src/scene_renderer.cpp src/text_editor_widget.h)
target_include_directories(callayers_renderer PUBLIC src PRIVATE "${ULTRALIGHT_ROOT}/include")
target_link_directories(callayers_renderer PUBLIC "${ULTRALIGHT_ROOT}/bin")
target_link_libraries(callayers_renderer PUBLIC Qt6::Widgets Ultralight AppCore WebCore UltralightCore)
file(COPY "${ULTRALIGHT_ROOT}/resources" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../web" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/../spec-v2/assets/callayers.png" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/assets")
option(CALLAYERS_DEVELOPMENT "Enable explicit local-testing flags" OFF)
set(CALLAYERS_SOURCES
  src/auth_callback.cpp
  src/capture.cpp
  src/output.cpp
  src/compose.cpp
  src/store.cpp
  src/secret.cpp
  src/auth.cpp
  src/telemetry.cpp
  src/preview.cpp
  src/window.cpp)
add_executable(callayers src/main.cpp ${CALLAYERS_SOURCES})
target_include_directories(callayers PRIVATE src)
target_link_libraries(callayers PRIVATE Qt6::Widgets Qt6::Network Qt6::DBus callayers_renderer)
set_target_properties(callayers PROPERTIES INSTALL_RPATH "$ORIGIN/../lib/callayers")
target_compile_definitions(callayers PRIVATE "CALLAYERS_VERSION=\"${PROJECT_VERSION}\"")
if(CALLAYERS_DEVELOPMENT)
  target_compile_definitions(callayers PRIVATE CALLAYERS_DEVELOPMENT)
endif()
install(TARGETS callayers RUNTIME DESTINATION bin)
install(DIRECTORY "${ULTRALIGHT_ROOT}/bin/" DESTINATION lib/callayers FILES_MATCHING PATTERN "*.so*")
install(DIRECTORY "${ULTRALIGHT_ROOT}/resources" "${CMAKE_CURRENT_SOURCE_DIR}/../web" DESTINATION share/callayers)
install(DIRECTORY "${ULTRALIGHT_ROOT}/license/" DESTINATION share/doc/callayers/ultralight)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../spec-v2/assets/callayers.png" DESTINATION share/callayers/assets)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../spec-v2/assets/callayers-mark.svg" DESTINATION share/icons/hicolor/scalable/apps RENAME callayers.svg)
install(FILES packaging/callayers.desktop DESTINATION share/applications)
include(CTest)
if(BUILD_TESTING)
  add_executable(linux_renderer_checks tests/renderer_checks.cpp)
  target_link_libraries(linux_renderer_checks PRIVATE callayers_renderer)
  add_test(NAME linux_html_renderer COMMAND linux_renderer_checks)
  set_tests_properties(linux_html_renderer PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(linux_checks tests/checks.cpp src/auth_callback.cpp)
  target_include_directories(linux_checks PRIVATE src)
  add_test(NAME linux_video_and_callback COMMAND linux_checks)
  add_executable(linux_composition_checks tests/composition_checks.cpp src/compose.cpp)
  target_include_directories(linux_composition_checks PRIVATE src)
  target_link_libraries(linux_composition_checks PRIVATE Qt6::Gui)
  add_test(NAME linux_composition COMMAND linux_composition_checks)
  add_executable(linux_camera_checks tests/camera_checks.cpp src/capture.cpp)
  target_include_directories(linux_camera_checks PRIVATE src)
  target_link_libraries(linux_camera_checks PRIVATE Qt6::Widgets)
  add_test(NAME linux_camera_selection_and_recovery COMMAND linux_camera_checks)
  set_tests_properties(linux_camera_selection_and_recovery PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(CALLAYERS_DEVELOPMENT)
    add_executable(linux_parity_checks tests/parity_checks.cpp ${CALLAYERS_SOURCES})
    target_include_directories(linux_parity_checks PRIVATE src)
    target_link_libraries(linux_parity_checks PRIVATE Qt6::Widgets Qt6::Network Qt6::DBus callayers_renderer)
    target_compile_definitions(linux_parity_checks PRIVATE CALLAYERS_DEVELOPMENT)
    add_test(NAME linux_preview_interactions COMMAND linux_parity_checks --offline-preview --skip-sign-in)
    set_tests_properties(linux_preview_interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  endif()
  add_executable(linux_workspace_checks tests/workspace_checks.cpp ${CALLAYERS_SOURCES})
  target_include_directories(linux_workspace_checks PRIVATE src)
  target_link_libraries(linux_workspace_checks PRIVATE Qt6::Widgets Qt6::Network Qt6::DBus callayers_renderer)
  if(CALLAYERS_DEVELOPMENT)
    target_compile_definitions(linux_workspace_checks PRIVATE CALLAYERS_DEVELOPMENT)
    add_test(NAME linux_development_workspace COMMAND linux_workspace_checks --skip-sign-in)
    set_tests_properties(linux_development_workspace PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
    add_test(NAME linux_sign_in_gate COMMAND linux_workspace_checks --offline-preview)
    set_tests_properties(linux_sign_in_gate PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;DBUS_SESSION_BUS_ADDRESS=unix:path=/tmp/callayers-test-no-session-bus" TIMEOUT 20)
  else()
    add_test(NAME linux_production_sign_in_gate COMMAND linux_workspace_checks --skip-sign-in --offline-preview)
    set_tests_properties(linux_production_sign_in_gate PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;DBUS_SESSION_BUS_ADDRESS=unix:path=/tmp/callayers-test-no-session-bus" TIMEOUT 20)
  endif()
  add_test(NAME linux_install_check COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/packaging/install.sh" --check)
endif()
